// Chat Messages for escalation system
model ChatMessage {
  id            String    @id @default(cuid())
  userId        Int?      // NULL if guest not logged in when asking
  guestEmail    String?   // Capture email if guest provides
  message       String    @db.Text
  senderType    String    @default("guest") // "guest" | "bot" | "admin"
  
  // Bot response info
  isBotMatched  Boolean   @default(false)
  botResponse   String?   @db.Text
  
  // Escalation fields
  isEscalated   Boolean   @default(false)
  escalatedAt   DateTime?
  
  // Admin reply
  adminReply    String?   @db.Text
  resolvedBy    Int?      // admin user ID
  resolvedAt    DateTime?
  
  // Metadata
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  // Relationships
  user          User?     @relation(fields: [userId], references: [id])
  
  @@index([userId])
  @@index([isEscalated])
  @@index([createdAt])
}

// Chat Escalation Requests (when guest clicks "Request Admin Contact")
model ChatEscalation {
  id            String    @id @default(cuid())
  userId        Int       // ONLY logged-in guests can escalate
  guestEmail    String
  reason        String    @db.Text  // Why guest needs admin contact
  status        String    @default("pending") // pending | contacted | resolved
  contactedAt   DateTime?
  contactedBy   Int?      // admin user ID who responded
  notes         String?   @db.Text  // Admin notes
  
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt
  
  // Relationships
  user          User      @relation(fields: [userId], references: [id])
  contactedByUser User?   @relation("EscalationContactedBy", fields: [contactedBy], references: [id])
  
  @@index([userId])
  @@index([status])
  @@index([createdAt])
}
